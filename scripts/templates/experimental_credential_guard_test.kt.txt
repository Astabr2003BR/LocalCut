package com.localcut.editor.bridge.agent

import com.localcut.editor.bridge.security.BridgeCredentialStore
import com.localcut.editor.bridge.security.BridgeStoredCredential
import com.localcut.editor.bridge.security.ExperimentalCredentialRead
import org.junit.Assert.*
import org.junit.Test

class ExperimentalCredentialGuardTest {
    private val credential = BridgeStoredCredential("a", "r", 10, "recovery")
    private fun read(fields: Map<String, String?>, corrupt: Boolean = false): BridgeStoredCredential? =
        ExperimentalCredentialRead.load(
            { fields.containsKey("iv") }, { fields.containsKey("ciphertext") },
            { fields["iv"] }, { fields["ciphertext"] },
            { _, _ -> if (corrupt) throw IllegalArgumentException("secret") else credential },
        )
    private fun store(fields: MutableMap<String, String?>, corrupt: Boolean = false) = object : BridgeCredentialStore {
        override fun load(): BridgeStoredCredential? = read(fields, corrupt)
        override fun save(credential: BridgeStoredCredential) { error("must not save") }
        override fun clear() { error("must not clear") }
    }
    @Test fun absentRecordCanChangeEndpoint() {
        val fields = mutableMapOf<String, String?>()
        val settings = InMemoryBridgeSettingsStore(relayBaseUrl = "https://one.example")
        assertTrue(BridgeEndpointPreference(settings, store(fields)).save("https://two.example").isSuccess)
        assertEquals("https://two.example", settings.relayBaseUrl)
        assertTrue(fields.isEmpty())
    }
    @Test fun intactRecordBlocksChange() {
        val fields = mutableMapOf<String, String?>("iv" to "iv", "ciphertext" to "encrypted")
        val snapshot = fields.toMap()
        assertEquals(credential, read(fields))
        val settings = InMemoryBridgeSettingsStore(relayBaseUrl = "https://one.example")
        assertTrue(BridgeEndpointPreference(settings, store(fields)).save("https://two.example").isFailure)
        assertEquals("https://one.example", settings.relayBaseUrl)
        assertEquals(snapshot, fields)
    }
    @Test fun incompleteNullAndCorruptRecordsBlockAndArePreserved() {
        val cases = listOf(
            mutableMapOf<String, String?>("iv" to "iv"),
            mutableMapOf<String, String?>("ciphertext" to "encrypted"),
            mutableMapOf<String, String?>("iv" to null, "ciphertext" to "encrypted"),
            mutableMapOf<String, String?>("iv" to "iv", "ciphertext" to "encrypted"),
        )
        cases.forEachIndexed { index, fields ->
            val before = fields.toMap()
            val settings = InMemoryBridgeSettingsStore(relayBaseUrl = "https://one.example")
            val result = BridgeEndpointPreference(settings, store(fields, index == 3)).save("https://two.example")
            assertEquals("PAIRING_STATE_UNKNOWN", result.exceptionOrNull()?.message)
            assertEquals("https://one.example", settings.relayBaseUrl)
            assertEquals(before, fields)
        }
    }
    @Test fun sameEndpointIsNoOpEvenWhenReadFails() {
        val fields = mutableMapOf<String, String?>("iv" to "iv")
        val settings = InMemoryBridgeSettingsStore(relayBaseUrl = "https://one.example")
        assertTrue(BridgeEndpointPreference(settings, store(fields)).save("https://ONE.example/").isSuccess)
        assertEquals(mapOf("iv" to "iv"), fields)
    }
    @Test fun preferenceAndDecryptFailuresAreUnknown() {
        val failure = runCatching {
            ExperimentalCredentialRead.load<String>(
                { throw IllegalStateException("private") }, { false }, { null }, { null }, { _, _ -> "" },
            )
        }.exceptionOrNull()
        assertEquals("PAIRING_STATE_UNKNOWN", failure?.message)
        assertNull(failure?.cause)
    }
    @Test fun backingFileOrReadFailureIsUnknown() {
        for (backing in listOf<() -> Boolean>({ true }, { throw IllegalStateException("unavailable") })) {
            val failure = runCatching {
                ExperimentalCredentialRead.load<String>(
                    { false }, { false }, { null }, { null }, { _, _ -> "" }, backing,
                )
            }.exceptionOrNull()
            assertEquals("PAIRING_STATE_UNKNOWN", failure?.message)
        }
    }
}
