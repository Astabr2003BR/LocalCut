    override fun load(): BridgeStoredCredential? = ExperimentalCredentialRead.load(
        hasIv = { preferences.contains(KEY_IV) },
        hasCiphertext = { preferences.contains(KEY_CIPHERTEXT) },
        readIv = { preferences.getString(KEY_IV, null) },
        readCiphertext = { preferences.getString(KEY_CIPHERTEXT, null) },
        backingFileExists = {
            // list() returning null is an unknown state, never proof of absence.
            val appDir = checkNotNull(preferencesDirectory.parentFile)
            val appEntries = checkNotNull(appDir.list())
            if (!appEntries.contains("shared_prefs")) {
                false
            } else {
                val entries = checkNotNull(preferencesDirectory.list())
                entries.contains("$PREFERENCES_NAME.xml") ||
                    entries.contains("$PREFERENCES_NAME.xml.bak")
            }
        },
        decrypt = { ivEncoded, ciphertextEncoded ->
            val iv = Base64.getDecoder().decode(ivEncoded)
            val ciphertext = Base64.getDecoder().decode(ciphertextEncoded)
            val keyStore = KeyStore.getInstance(ANDROID_KEYSTORE).apply { load(null) }
            val key = keyStore.getKey(CREDENTIAL_KEY_ALIAS, null) as? SecretKey
                ?: throw IllegalStateException("CREDENTIAL_KEY_UNAVAILABLE")
            val cipher = Cipher.getInstance(TRANSFORMATION)
            cipher.init(Cipher.DECRYPT_MODE, key, GCMParameterSpec(GCM_TAG_BITS, iv))
            decode(cipher.doFinal(ciphertext))
        },
    )
