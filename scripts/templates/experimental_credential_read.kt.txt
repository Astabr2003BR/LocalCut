package com.localcut.editor.bridge.security

/** A null result means both fields were verified absent, not unreadable. */
internal object ExperimentalCredentialRead {
    fun <T : Any> load(
        hasIv: () -> Boolean,
        hasCiphertext: () -> Boolean,
        readIv: () -> String?,
        readCiphertext: () -> String?,
        decrypt: (String, String) -> T,
        backingFileExists: () -> Boolean = { false },
    ): T? = try {
        val ivPresent = hasIv()
        val ciphertextPresent = hasCiphertext()
        if (!ivPresent && !ciphertextPresent) {
            check(!backingFileExists())
            null
        }
        else {
            check(ivPresent && ciphertextPresent)
            val iv = checkNotNull(readIv())
            val ciphertext = checkNotNull(readCiphertext())
            decrypt(iv, ciphertext)
        }
    } catch (_: Exception) {
        // No data or exception details are exposed; never delete or replace data.
        throw IllegalStateException("PAIRING_STATE_UNKNOWN")
    }
}
